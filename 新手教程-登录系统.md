### 3.3 编写用户模型（models/user.go）

> **模型（Model）** 就是把数据库中的"表"映射成 Go 代码中的"结构体"，我们用它来操作数据。

创建 `models/user.go`，内容如下：

```go
package models

import (
	"errors"
	"golang.org/x/crypto/bcrypt"
	"gorm.io/driver/sqlite"
	"gorm.io/gorm"
)

// DB 是全局数据库连接对象，其他文件都通过它来操作数据库
var DB *gorm.DB

// User 对应数据库里的 users 表
// 每一行注释是 GORM 的标签，用来定义字段约束
type User struct {
	gorm.Model                // 内嵌：自动加 ID、CreatedAt、UpdatedAt、DeletedAt
	Username string `gorm:"uniqueIndex;size:50;not null" json:"username"`  // 用户名：唯一索引，最多50字符
	Password string `gorm:"size:255;not null" json:"-"`                    // 密码：json:"-" 表示API返回时不返回密码（安全！）
	Email    string `gorm:"size:100" json:"email"`                         // 邮箱：可选
}

// RegisterRequest 注册接口的入参结构
type RegisterRequest struct {
	Username string `json:"username" binding:"required,min=3,max=50"`  // binding 标签做参数校验
	Password string `json:"password" binding:"required,min=6"`
	Email    string `json:"email"`
}

// LoginRequest 登录接口的入参结构
type LoginRequest struct {
	Username string `json:"username" binding:"required"`
	Password string `json:"password" binding:"required"`
}

// InitDB 初始化数据库连接，自动建表
func InitDB() error {
	var err error
	// 打开 SQLite 数据库文件（如果不存在会自动创建）
	DB, err = gorm.Open(sqlite.Open("users.db"), &gorm.Config{})
	if err != nil {
		return err
	}
	// AutoMigrate：自动根据 User 结构体建表（已有表会自动加新列）
	return DB.AutoMigrate(&User{})
}

// HashPassword 把明文密码变成哈希值存储（永远不要存明文密码！）
func (u *User) HashPassword() error {
	hashed, err := bcrypt.GenerateFromPassword([]byte(u.Password), bcrypt.DefaultCost)
	if err != nil {
		return err
	}
	u.Password = string(hashed)
	return nil
}

// CheckPassword 登录时验证密码是否正确
func (u *User) CheckPassword(password string) bool {
	err := bcrypt.CompareHashAndPassword([]byte(u.Password), []byte(password))
	return err == nil
}

// CreateUser 创建一个新用户
func CreateUser(user *User) error {
	if user.Username == "" || user.Password == "" {
		return errors.New("用户名和密码不能为空")
	}
	// 先检查用户名是否已存在
	var count int64
	DB.Model(&User{}).Where("username = ?", user.Username).Count(&count)
	if count > 0 {
		return errors.New("用户名已存在")
	}
	return DB.Create(user).Error
}

// FindUserByUsername 根据用户名查找用户（登录时用）
func FindUserByUsername(username string) (*User, error) {
	var user User
	result := DB.Where("username = ?", username).First(&user)
	if result.Error != nil {
		return nil, result.Error
	}
	return &user, nil
}

// FindUserByID 根据 ID 查找用户（获取个人信息时用）
func FindUserByID(id uint) (*User, error) {
	var user User
	result := DB.First(&user, id)
	if result.Error != nil {
		return nil, result.Error
	}
	return &user, nil
}
```

> 🔑 **Bcrypt 加密的核心思想**：
> 即使数据库泄露，黑客也无法还原出用户的真实密码。每次加密同一个密码，结果也不一样（加盐）。

### 3.4 编写 JWT 中间件（middleware/auth.go）

> **JWT（JSON Web Token）** 是登录认证的标准方案。用户登录成功后，服务端签发一个加密的"令牌"，之后的请求都要带上这个令牌来证明"我已登录"。

创建 `middleware/auth.go`：

```go
package middleware

import (
	"net/http"
	"strings"
	"time"
	"github.com/gin-gonic/gin"
	"github.com/golang-jwt/jwt/v5"
	"login-backend/models"
)

// JWTSecret 签名密钥：生产环境务必换成复杂的随机字符串！
// ⚠️ 泄露这个密钥，任何人都能伪造任意用户的令牌
var JWTSecret = []byte("your-secret-key-change-in-production")

// TokenExpire Token 有效期：24小时
const TokenExpire = time.Hour * 24

// Claims 是 Token 中携带的自定义数据
type Claims struct {
	UserID   uint   `json:"user_id"`   // 用户 ID
	Username string `json:"username"`  // 用户名
	jwt.RegisteredClaims               // 内嵌：过期时间、签发时间等标准字段
}

// GenerateToken 登录/注册成功后，生成 Token 返回给前端
func GenerateToken(user *models.User) (string, error) {
	claims := Claims{
		UserID:   user.ID,
		Username: user.Username,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(TokenExpire)),  // 过期时间 = 当前时间 + 24h
			IssuedAt:  jwt.NewNumericDate(time.Now()),                   // 签发时间
		},
	}
	// 使用 HS256 算法签名
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	// 用密钥签名后得到字符串 Token
	return token.SignedString(JWTSecret)
}

// AuthMiddleware 是一个 Gin 中间件，用于保护需要登录的接口
// 比如：获取用户信息、修改资料等
func AuthMiddleware() gin.HandlerFunc {
	return func(c *gin.Context) {
		// 1. 从请求头获取 Authorization
		authHeader := c.GetHeader("Authorization")
		if authHeader == "" {
			c.JSON(http.StatusUnauthorized, gin.H{"error": "缺少认证令牌"})
			c.Abort()  // 终止后续处理
			return
		}

		// 2. 格式校验：必须是 "Bearer xxx.yyy.zzz"
		parts := strings.Split(authHeader, " ")
		if len(parts) != 2 || parts[0] != "Bearer" {
			c.JSON(http.StatusUnauthorized, gin.H{"error": "认证令牌格式错误"})
			c.Abort()
			return
		}

		// 3. 解析并验证 Token
		tokenStr := parts[1]
		claims := &Claims{}
		token, err := jwt.ParseWithClaims(tokenStr, claims, func(token *jwt.Token) (interface{}, error) {
			return JWTSecret, nil  // 用同一个密钥验签
		})

		if err != nil || !token.Valid {
			c.JSON(http.StatusUnauthorized, gin.H{"error": "令牌无效或已过期"})
			c.Abort()
			return
		}

		// 4. Token 合法！把用户信息存到请求上下文，后面的处理器可以直接用
		c.Set("user_id", claims.UserID)
		c.Set("username", claims.Username)
		c.Next()  // 放行，继续执行后面的处理器
	}
}
```

### 3.5 编写接口处理器（handlers/auth.go）

> **处理器（Handler）** 就是每个 API 接口的具体逻辑：接收请求 → 处理 → 返回响应。

创建 `handlers/auth.go`：

```go
package handlers

import (
	"net/http"
	"github.com/gin-gonic/gin"
	"login-backend/middleware"
	"login-backend/models"
)

// Register POST /api/auth/register
func Register(c *gin.Context) {
	// 1. 绑定并校验请求参数
	var req models.RegisterRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "请求参数错误: " + err.Error()})
		return
	}

	// 2. 构造 User 对象
	user := &models.User{
		Username: req.Username,
		Password: req.Password,
		Email:    req.Email,
	}

	// 3. 密码加密（一定要加密！）
	if err := user.HashPassword(); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "密码加密失败"})
		return
	}

	// 4. 写入数据库
	if err := models.CreateUser(user); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	// 5. 注册即登录：签发 Token
	token, err := middleware.GenerateToken(user)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "生成令牌失败"})
		return
	}

	// 6. 返回注册成功（201 Created）
	c.JSON(http.StatusCreated, gin.H{
		"message": "注册成功",
		"token":   token,
		"user": gin.H{
			"id":       user.ID,
			"username": user.Username,
			"email":    user.Email,
		},
	})
}

// Login POST /api/auth/login
func Login(c *gin.Context) {
	// 1. 绑定参数
	var req models.LoginRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "请求参数错误: " + err.Error()})
		return
	}

	// 2. 查找用户
	user, err := models.FindUserByUsername(req.Username)
	if err != nil {
		// ⚠️ 安全提示：不要说"用户名不存在"，避免攻击者枚举用户名
		c.JSON(http.StatusUnauthorized, gin.H{"error": "用户名或密码错误"})
		return
	}

	// 3. 验证密码
	if !user.CheckPassword(req.Password) {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "用户名或密码错误"})
		return
	}

	// 4. 签发 Token
	token, err := middleware.GenerateToken(user)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "生成令牌失败"})
		return
	}

	// 5. 返回登录成功
	c.JSON(http.StatusOK, gin.H{
		"message": "登录成功",
		"token":   token,
		"user": gin.H{
			"id":       user.ID,
			"username": user.Username,
			"email":    user.Email,
		},
	})
}

// GetProfile GET /api/user/profile  （受保护的接口）
func GetProfile(c *gin.Context) {
	// 从中间件设置的上下文中取出 user_id
	userID, _ := c.Get("user_id")
	user, err := models.FindUserByID(userID.(uint))
	if err != nil {
		c.JSON(http.StatusNotFound, gin.H{"error": "用户不存在"})
		return
	}

	c.JSON(http.StatusOK, gin.H{
		"user": gin.H{
			"id":        user.ID,
			"username":  user.Username,
			"email":     user.Email,
			"createdAt": user.CreatedAt,
		},
	})
}
```

### 3.6 编写入口文件 main.go

> 这是程序的启动入口，把上面所有模块组装起来：初始化数据库 → 配置 CORS → 注册路由 → 启动服务器。

创建 `main.go`：

```go
package main

import (
	"log"
	"github.com/gin-contrib/cors"
	"github.com/gin-gonic/gin"
	"login-backend/handlers"
	"login-backend/middleware"
	"login-backend/models"
)

func main() {
	// 1. 初始化数据库
	if err := models.InitDB(); err != nil {
		log.Fatal("数据库初始化失败:", err)
	}
	log.Println("数据库连接成功")

	// 2. 创建 Gin 实例（默认带 Logger 和 Recovery 中间件）
	r := gin.Default()

	// 3. 配置 CORS（跨域）：允许前端 5173/3000 端口访问
	// 前后端分离开发必须配置，否则浏览器会拦截请求
	r.Use(cors.New(cors.Config{
		AllowOrigins:     []string{"http://localhost:5173", "http://localhost:3000"},
		AllowMethods:     []string{"GET", "POST", "PUT", "DELETE", "OPTIONS"},
		AllowHeaders:     []string{"Origin", "Content-Type", "Authorization"},
		ExposeHeaders:    []string{"Content-Length"},
		AllowCredentials: true,
	}))

	// 4. 注册路由
	api := r.Group("/api")
	{
		// 认证相关接口（公开，不需要登录）
		auth := api.Group("/auth")
		{
			auth.POST("/register", handlers.Register)
			auth.POST("/login", handlers.Login)
		}

		// 用户相关接口（受保护，必须加 AuthMiddleware）
		user := api.Group("/user")
		user.Use(middleware.AuthMiddleware())  // 这一行给整个 group 加了登录保护
		{
			user.GET("/profile", handlers.GetProfile)
		}
	}

	// 5. 健康检查接口（验证服务器是否正常运行）
	r.GET("/health", func(c *gin.Context) {
		c.JSON(200, gin.H{"status": "ok"})
	})

	// 6. 启动服务器，监听 8080 端口
	log.Println("服务器启动在 :8080")
	if err := r.Run(":8080"); err != nil {
		log.Fatal("服务器启动失败:", err)
	}
}
```

### 4.4 API 服务封装（services/api.js）

> 统一把所有 HTTP 请求封装到一个文件里，方便维护。

创建 `src/services/api.js`：

```javascript
const BASE_URL = '/api'

// 通用请求函数：自动加 Token、自动解析 JSON、自动抛错
async function request(url, options = {}) {
  // 从 localStorage 取 Token
  const token = localStorage.getItem('token')
  const headers = {
    'Content-Type': 'application/json',
    ...options.headers,
  }
  if (token) {
    // 带上 Bearer Token
    headers['Authorization'] = `Bearer ${token}`
  }

  const response = await fetch(`${BASE_URL}${url}`, {
    ...options,
    headers,
  })

  const data = await response.json()
  // 非 2xx 状态码统一抛错，页面里 catch 处理
  if (!response.ok) {
    throw new Error(data.error || '请求失败')
  }
  return data
}

// 导出具体的 API
export const api = {
  // 注册
  register: (username, password, email) =>
    request('/auth/register', {
      method: 'POST',
      body: JSON.stringify({ username, password, email }),
    }),

  // 登录
  login: (username, password) =>
    request('/auth/login', {
      method: 'POST',
      body: JSON.stringify({ username, password }),
    }),

  // 获取个人信息（需要登录）
  getProfile: () => request('/user/profile'),
}
```

### 4.5 全局登录状态（context/AuthContext.jsx）

> **React Context** 可以把状态（比如"当前登录用户"）放到全局，任何组件都能读取，不用一层层传递 props。

创建 `src/context/AuthContext.jsx`：

```jsx
import { createContext, useContext, useState, useEffect } from 'react'
import { api } from '../services/api.js'

// 1. 创建 Context 容器
const AuthContext = createContext(null)

// 2. AuthProvider：放在根组件外层，管理登录状态
export function AuthProvider({ children }) {
  const [user, setUser] = useState(null)        // 当前用户信息
  const [token, setToken] = useState(localStorage.getItem('token'))  // Token
  const [loading, setLoading] = useState(true)  // 是否正在加载

  // 3. 页面刷新时，如果 localStorage 里有 Token，自动获取用户信息
  useEffect(() => {
    if (token) {
      loadProfile()
    } else {
      setLoading(false)
    }
  }, [token])

  async function loadProfile() {
    try {
      const data = await api.getProfile()
      setUser(data.user)
    } catch (error) {
      // Token 过期或无效 → 退出登录
      logout()
    } finally {
      setLoading(false)
    }
  }

  // 登录
  async function login(username, password) {
    const data = await api.login(username, password)
    localStorage.setItem('token', data.token)  // 存 Token 到本地
    setToken(data.token)
    setUser(data.user)
    return data
  }

  // 注册
  async function register(username, password, email) {
    const data = await api.register(username, password, email)
    localStorage.setItem('token', data.token)
    setToken(data.token)
    setUser(data.user)
    return data
  }

  // 退出登录
  function logout() {
    localStorage.removeItem('token')
    setToken(null)
    setUser(null)
  }

  // 4. 把状态和方法通过 Provider 传给所有子组件
  return (
    <AuthContext.Provider value={{ user, token, login, register, logout, loading }}>
      {children}
    </AuthContext.Provider>
  )
}

// 5. 自定义 Hook：任何组件调用 useAuth() 就能拿到登录状态
export function useAuth() {
  const context = useContext(AuthContext)
  if (!context) {
    throw new Error('useAuth 必须在 AuthProvider 内使用')
  }
  return context
}
```

> 流程图：
> ```
> 页面刷新
>   → 检查 localStorage 有没有 token
>     → 有：请求 /user/profile 拿用户信息（成功=已登录，失败=清token）
>     → 没有：就是未登录状态
> ```

## 6. 核心原理解释

### 6.1 登录认证的完整流程

```
          用户                     前端                     后端
           │  1. 输入账号密码        │                        │
           │──────────────────────>│  2. POST /api/auth/login│
           │                       │───────────────────────>│
           │                       │                        │ 3. 查数据库 + Bcrypt 验证密码
           │                       │                        │ 4. 生成 JWT Token（签名）
           │                       │  5. 返回 {token, user}  │
           │                       │<───────────────────────│
           │  6. 存 token 到 localStorage                   │
           │                       │                        │
           │  7. 访问受保护页面     │                        │
           │──────────────────────>│  8. GET /api/user/profile（带 Token）
           │                       │───────────────────────>│
           │                       │                        │ 9. JWT 中间件验签
           │                       │  10. 返回用户信息       │
           │                       │<───────────────────────│
           │  11. 渲染页面          │                        │
```

### 6.2 JWT Token 长什么样？

一个典型的 JWT 由三段组成，用 `.` 分隔：

```
eyJhbGciOiJIUzI1NiJ9.eyJ1c2VyX2lkIjoxLCJ1c2VybmFtZSI6InRlc3QiLCJleHAiOjE3MDAwMDAwMDB9.abc123signature
└────────────────────────┘ └──────────────────────────────────────────────────────┘ └────────────────────┘
          Header                            Payload（可以解码看到用户ID等）               签名（防篡改）
```

> 🔐 注意：Payload 只是 Base64 编码，**不是加密**，任何人都能解码查看。所以**不要在 JWT 里放密码、身份证号等敏感信息**。

### 6.3 为什么用 Bcrypt 不存明文？

| 存储方式 | 数据库泄露后果 |
|---------|-------------|
| 明文密码 | 黑客直接拿到所有用户的真实密码（99%的用户会在多网站复用同一个密码！后果极严重） |
| Bcrypt 哈希 | 黑客只能看到哈希值，无法还原出原密码 |

> Bcrypt 的特点：每次加密结果都不一样，验证时自动匹配。

### 6.4 CORS（跨域）是什么？

- 前端在 `http://localhost:5173`（源A）
- 后端在 `http://localhost:8080`（源B）

因为**端口不一样**，就叫"跨域"。浏览器出于安全默认会拦截这种请求，除非后端在响应头里明确说"我允许 5173 来访问我"。

所以后端配置了 CORS 中间件就是为了解决这个问题。

## 7. 常见问题 FAQ

### Q1：运行 `go run main.go` 报 `no required module provides package`？
A：先执行 `go mod tidy` 自动下载缺失的依赖。

### Q2：前端请求报 404 或 Proxy error？
A：检查后端是否启动（先确保访问 http://localhost:8080/health 返回 ok），然后重启前端 npm run dev。

### Q3：报 `sqlite: missing "?"` 之类的数据库错误？
A：删除 `backend/users.db` 文件（会自动重建），重启后端。

### Q4：登录成功但刷新页面又要重新登录？
A：检查 JWTSecret 是否一致。如果修改了密钥，之前签发的所有 Token 都会失效。

### Q5：如何切换到 MySQL 数据库？
A：
1. `go get gorm.io/driver/mysql`
2. 把 models/user.go 中的 `sqlite.Open("users.db")` 换成：
   ```go
   mysql.Open("user:password@tcp(127.0.0.1:3306)/login_db?charset=utf8mb4&parseTime=True&loc=Local")
   ```
3. 提前在 MySQL 中创建 `login_db` 数据库。

### Q6：生产环境要做哪些改进？
| 改进点 | 说明 |
|-------|------|
| 🔑 JWTSecret | 放到环境变量，不要硬编码 |
| 🛡️ HTTPS | 必须使用 HTTPS，防止 Token 被拦截 |
| 📝 日志 | 记录登录失败日志，做 IP 限流防暴力破解 |
| 🔐 密码策略 | 强制强密码、启用 2FA 双因素认证 |
| 🚫 Token 黑名单 | 支持"提前退出"让 Token 立即失效（Redis） |
| 🔄 Refresh Token | 双 Token 机制，缩短 Access Token 有效期 |
